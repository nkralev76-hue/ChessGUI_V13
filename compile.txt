:: Portable (works on all machines) — use for distribution
gcc -O3 -DUSE_BOOK -o chess_gui_v13.exe chess_gui_v13.c -lSDL2 -lSDL2_image -lm -lcomdlg32 -mwindows

:: Native optimized (your machine only, AVX2) — ~5-10% faster but not portable
gcc -O3 -march=native -mpopcnt -mavx2 -DUSE_BOOK -o chess_gui_v13.exe chess_gui_v13.c -lSDL2 -lSDL2_image -lm -lcomdlg32 -mwindows
